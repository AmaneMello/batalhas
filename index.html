function executarGiroLootingAndar() { let faixa = document.getElementById('loot-andar-faixa').value; let kaijusDerrotados = parseInt(document.getElementById('loot-andar-kaijus').value) || 0; let resultadoDiv = document.getElementById('resultado-looting-andar'); let manual = "Sem manual", nivelEnchimento = "Lv1", nivelKaijumita = "Lv1"; let minQtd = 1, maxQtd = 3; // Definindo parâmetros base por faixa de andar if (faixa === "20-15") { manual = "Sem manual"; nivelEnchimento = "Lv1"; nivelKaijumita = "Lv1"; minQtd = 1; maxQtd = 3; } else if (faixa === "15-10") { manual = "Lv1"; nivelEnchimento = "Lv1~2"; nivelKaijumita = "Lv1"; minQtd = 1; maxQtd = 4; } else if (faixa === "10-5") { manual = "Lv2"; nivelEnchimento = "Lv2"; nivelKaijumita = "Lv2"; minQtd = 1; maxQtd = 3; } else if (faixa === "5-terreo") { manual = "Lv2~3"; nivelEnchimento = "Lv2~3"; nivelKaijumita = "Lv2~3"; minQtd = 1; maxQtd = 4; } else if (faixa === "terreo-subsolo") { manual = "Lv3"; nivelEnchimento = "Lv3"; nivelKaijumita = "Lv3"; minQtd = 3; maxQtd = 5; } // Sorteia a quantidade exata dentro do intervalo da faixa let quantidadeSorteada = Math.floor(Math.random() * (maxQtd - minQtd + 1)) + minQtd; // Sorteia a quantidade específica de Kaijumita obtida na rolagem let qtdKaijumitaSorteada = Math.floor(Math.random() * 2) + 1; let extrasExtras = 0; // Bônus se houver rolagens acima de 5 (estritamente 6 ou mais) if (kaijusDerrotados >= 6) { let excesso = kaijusDerrotados - 6 + 1; for (let i = 0; i < excesso; i++) { let rollExtra = Math.random() * 100; if (rollExtra < 20) extrasExtras++; } } resultadoDiv.innerHTML = `Looting de Andar Revelado (${faixa}):
` + `- Manual: ${manual}
` + `- Enchimento (Atk/Suport/Def): ${nivelEnchimento} (Qtd: ${quantidadeSorteada})
` + `- Kaijumita: ${nivelKaijumita} (Qtd: ${qtdKaijumitaSorteada})
` + (extrasExtras > 0 ? `🔥 Bônus por Kaijus (Acima de 5): +${extrasExtras} item(ns) extra(s)!` : ''); }
